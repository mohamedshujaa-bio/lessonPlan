        extractedText = window.XLSX.utils.sheet_to_txt(worksheet);

      else if (fileExtension === 'pdf') {
        if (!window.pdfjsLib) {
          throw new Error("PDF processor is loading. Please try again in a moment.");
        }
        const arrayBuffer = await file.arrayBuffer();
        window.pdfjsLib.GlobalWorkerOptions.workerSrc = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/2.16.105/pdf.worker.min.js';
        const pdf = await window.pdfjsLib.getDocument({ data: arrayBuffer }).promise;

          const page = await pdf.getPage(i);
          const textContent = await page.getTextContent();
          const pageText = textContent.items.map(item => item.str).join(" ");
          textBuilder += `\n--- PAGE ${i} ---\n` + pageText;
        }
        extractedText = textBuilder;
      } 
      else {
        throw new Error("Unsupported file format. Please upload PDF, Word (.docx), Excel (.xlsx), CSV, or text (.txt).");
      }

      if (!extractedText || extractedText.trim().length === 0) {
        throw new Error("Could not extract text. Please ensure the file contains searchable digital text.");
      }

      setUploadedText(extractedText);
      setUploadedFileName(file.name);

      // Instantly run the targeted multi-lesson generator using target week input
      await triggerWeekAutoFill(weekInput);

      setErrorMsg(err.message || "An error occurred while uploading & parsing the file.");

                          <summary className="font-bold text-[10px] text-teal-700 cursor-pointer hover:text-teal-600 flex items-center gap-1">
                            <Eye className="w-3.5 h-3.5" /> Toggle Marking Guide (Expected Answers)
                          </summary>
                          <table className="w-full text-[9px] border-collapse mt-3" style={{ border: '1px solid #999999' }}>
                            <tbody>
                              <tr>
                                <td colSpan={2} className="bg-green-fill text-center font-bold py-1 border border-gray-400" style={{ backgroundColor: '#b7d7a8' }}>Marking Guide & Solution Outlines</td>
                              </tr>
                              {les.worksheet.questions.map((q, qKey) => (
                                <tr key={qKey}>
                                  <td className="bg-green-fill font-bold p-1 text-center border border-gray-400" style={{ backgroundColor: '#b7d7a8', width: '15%' }}>Q {q.qNo} expected</td>
                                  <td className="p-1 border border-gray-400 bg-white leading-relaxed text-gray-800">{q.expectedAnswer}</td>
                                </tr>
                              ))}
                            </tbody>
                          </table>
                        </details>
                      </div>

                      <div className="absolute bottom-4 left-0 right-0 text-center text-[9px] text-gray-400 select-none">
                        Lesson Number {les.lessonNumber} — Page 3 of 3
                      </div>
                    </div>
                  )}

                </div>
              );
            })}

          </div>
        </section>
      </main>

      {/* Modern High-Fidelity Worksheet Generator Modal Overlay */}
      {isWorksheetModalOpen && (
        <div className="fixed inset-0 bg-black/75 backdrop-blur-sm flex items-center justify-center z-50 p-4 animate-fadeIn">
          <div className="bg-slate-900 border border-slate-800 rounded-xl max-w-md w-full p-6 space-y-4 shadow-2xl">
            {/* Header */}
            <div className="flex justify-between items-center pb-2 border-b border-slate-800">
              <h3 className="text-sm font-bold text-white uppercase tracking-wider flex items-center gap-2">
                <Sparkles className="w-4.5 h-4.5 text-emerald-400 animate-pulse" />
                AI Worksheet Generation Config
              </h3>
              <button 
                onClick={() => setIsWorksheetModalOpen(false)}
                className="text-slate-400 hover:text-white transition-colors text-xs font-semibold"
              >
                ✕ Close
              </button>
            </div>

            {/* Config Fields */}
            <div className="space-y-3 text-xs">
              <div>
                <label className="block text-[11px] text-slate-400 mb-1 font-medium">Select Target Student Grade</label>
                <select 
                  value={wsGrade}
                  onChange={(e) => setWsGrade(e.target.value)}
                  className="w-full bg-white border border-slate-300 rounded px-2.5 py-1.5 text-black focus:outline-none focus:ring-2 focus:ring-teal-500"
                >
                  {[...Array(12)].map((_, i) => (
                    <option key={i} value={"Grade " + (i + 1)}>Grade {i + 1}</option>
                  ))}
                </select>
              </div>

              <div>
                <label className="block text-[11px] text-emerald-400 font-semibold mb-1">Google Drive / Shared Link Context</label>
                <input 
                  type="text" 
                  placeholder="Paste shared syllabus folder or textbook URL..."
                  value={wsDriveLink || ""}
                  onChange={(e) => setWsDriveLink(e.target.value)}
                  className="w-full bg-white border border-slate-300 rounded px-2.5 py-1.5 text-black focus:outline-none focus:ring-2 focus:ring-teal-500"
                />
              </div>

              <div>
                <label className="block text-[11px] text-slate-400 mb-1">Worksheet Question Format</label>
                <select 
                  value={wsType}
                  onChange={(e) => setWsType(e.target.value)}
                  className="w-full bg-white border border-slate-300 rounded px-2.5 py-1.5 text-black focus:outline-none focus:ring-2 focus:ring-teal-500"
                >
                  <option value="Question and Answer">Question and Answer (Standard Open Analytical)</option>
                  <option value="MCQ">MCQ (Multiple Choice Questions with options A, B, C, D)</option>
                  <option value="Quiz">Quiz (Rapid retrieval conceptual checkpoints)</option>
                  <option value="Fill in the blanks">Fill in the Blanks (Contextual completion sentences)</option>
                  <option value="Match the columns">Match the Columns (Column A and B Pairings)</option>
                </select>
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-[11px] text-slate-400 mb-1">Amount of Questions</label>
                  <select 
                    value={wsNumQuestions}
                    onChange={(e) => setWsNumQuestions(e.target.value)}
                    className="w-full bg-white border border-slate-300 rounded px-2.5 py-1.5 text-black focus:outline-none focus:ring-2 focus:ring-teal-500"
                  >
                    <option value="3">3 Questions</option>
                    <option value="5">5 Questions</option>
                    <option value="10">10 Questions</option>
                    <option value="15">15 Questions (Full Exam)</option>
                  </select>
                </div>
                <div>
                  <label className="block text-[11px] text-slate-400 mb-1">Differentiated level</label>
                  <select 
                    value={wsStudentLevel}
                    onChange={(e) => setWsStudentLevel(e.target.value)}
                    className="w-full bg-white border border-slate-300 rounded px-2.5 py-1.5 text-black focus:outline-none focus:ring-2 focus:ring-teal-500"
                  >
                    <option value="Normal">Normal (Grade-standard conceptual)</option>
                    <option value="ICP">ICP (Highly structured with scaffolded guides)</option>
                    <option value="lower level">lower level (Recall based with simplified terms)</option>
                  </select>
                </div>
              </div>

              <div>
                <label className="block text-[11px] text-slate-400 mb-1">Duration Limit</label>
                <select 
                  value={wsTimeLimit}
                  onChange={(e) => setWsTimeLimit(e.target.value)}
                  className="w-full bg-white border border-slate-300 rounded px-2.5 py-1.5 text-black focus:outline-none focus:ring-2 focus:ring-teal-500"
                >
                  <option value="10 mins">10 minutes</option>
                  <option value="15 mins">15 minutes</option>
                  <option value="20 mins">20 minutes</option>
                  <option value="30 mins">30 minutes</option>
                </select>
              </div>
            </div>

            {/* Error Message */}
            {errorMsg && (
              <div className="bg-red-950/40 border border-red-500/30 p-2.5 rounded-lg text-[11px] text-red-200 flex items-start gap-1.5">
                <AlertCircle className="w-4 h-4 shrink-0 mt-0.5" />
                <span>{errorMsg}</span>
              </div>
            )}

            {/* Footer Buttons */}
            <div className="flex justify-between items-center pt-2 gap-3 flex-wrap">
              {/* Export Active Worksheet PDF directly if it exists */}
              {activeLesson.worksheet ? (
                <button
                  type="button"
                  onClick={handleExportWorksheetPDF}
                  className="px-3.5 py-2 text-xs font-bold text-emerald-300 hover:text-white bg-emerald-950/40 hover:bg-emerald-900/60 border border-emerald-500/30 rounded-lg flex items-center gap-1.5 transition-colors"
                >
                  <Printer className="w-3.5 h-3.5" />
                  Export PDF Handout
                </button>
              ) : <div />}

              <div className="flex gap-2 ml-auto">
                <button
                  type="button"
                  onClick={() => setIsWorksheetModalOpen(false)}
                  className="px-4 py-2 text-xs font-semibold text-slate-300 hover:text-white bg-slate-800 rounded-lg hover:bg-slate-700 transition-colors"
                >
                  Cancel
                </button>
                <button
                  type="button"
                  onClick={async () => {
                    await generateClassWorksheet();
                    setIsWorksheetModalOpen(false);
                  }}
                  disabled={worksheetLoading}
                  className="px-4 py-2 text-xs font-bold text-white bg-gradient-to-r from-emerald-600 to-teal-600 rounded-lg flex items-center gap-1 hover:from-emerald-500 hover:to-teal-500 transition-all disabled:opacity-40"
                >
                  {worksheetLoading ? "Scouting..." : "✨ Generate Worksheet"}
                </button>
              </div>
            </div>
          </div>
        </div>
      )}

      {/* Styled inline Print Stylesheet */}
      <style>{`
        @media print {
          body {
            background: white !important;
            color: black !important;
          }
          header, section:first-of-type, .preview-container > div:first-child {
            display: none !important;
          }
          .preview-container {
            padding: 0 !important;
            margin: 0 !important;
            max-height: none !important;
            overflow: visible !important;
            display: block !important;
          }
          .a4-page {
            box-shadow: none !important;
            padding: 1.5cm 2cm !important;
            margin: 0 auto 30px auto !important;
            page-break-after: always !important;
            width: 21cm !important;
            height: 29.7cm !important;
            font-size: 10pt !important;
            border: none !important;
          }
          table {
            width: 100% !important;
          }
        }

        .a4-page {
          width: 21cm;
          min-height: 29.7cm;
          aspect-ratio: 1 / 1.414;
        }

        .bg-green-fill {
          background-color: #b7d7a8 !important;
        }

        @keyframes fadeIn {
          from { opacity: 0; transform: translateY(4px); }
          to { opacity: 1; transform: translateY(0); }
        }

        .animate-fadeIn {
          animation: fadeIn 0.15s ease-out forwards;
        }
      `}</style>
    </div>
  );
}
